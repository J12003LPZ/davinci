cmake_minimum_required(VERSION 3.16)
project(davinci_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(flag BUILD_SHARED_LIBS WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES
    WHISPER_BUILD_SERVER WHISPER_CURL WHISPER_SDL2 WHISPER_COREML
    GGML_NATIVE GGML_BACKEND_DL GGML_METAL GGML_ACCELERATE GGML_BLAS
    GGML_CUDA GGML_VULKAN GGML_OPENMP GGML_RPC GGML_LLAMAFILE GGML_CPU_KLEIDIAI
    GGML_SSE42 GGML_AVX GGML_AVX2 GGML_AVX_VNNI GGML_FMA GGML_F16C GGML_BMI2
    GGML_AVX512 GGML_AVX512_VBMI GGML_AVX512_VNNI GGML_AVX512_BF16
    GGML_AMX_TILE GGML_AMX_INT8 GGML_AMX_BF16 GGML_BUILD_TESTS GGML_BUILD_EXAMPLES)
    set(${flag} OFF CACHE BOOL "Portable local CPU voice build" FORCE)
endforeach()
option(DAVINCI_VOICE_NATIVE_CPU "Tune for this build machine; not for distribution" OFF)
set(GGML_NATIVE ${DAVINCI_VOICE_NATIVE_CPU} CACHE BOOL "Host CPU tuning" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
add_subdirectory(upstream)
add_library(davinci_voice_shim STATIC voice_shim.cpp)
target_link_libraries(davinci_voice_shim PRIVATE whisper)
install(TARGETS davinci_voice_shim whisper ggml ggml-base ggml-cpu ARCHIVE DESTINATION lib)
